  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("転記先シート");
  
  if (!sheet) {
    throw new Error("「転記先シート」が見つかりません。");
  }
  
  // フォームの入力内容を配列にしてシートの最終行に追加
  // ※実際の列の順番に合わせてformObjectのプロパティを並べてください
  const rowData = [
    formObject.name,      // A列: 名前 (sample)
    formObject.email,     // B列: メール (sample@nanndemoya.site)
    formObject.phone,     // C列: 電話番号 (7090758210)
    formObject.date,      // D列: 日付 (20260101)
    formObject.post       // E列: 郵便番号(3380837)
    formObject.house_name, // F列: 物件名(samplehouse)
    formObject.house_post, // G列: 郵便番号(3360035)
    fomrObject.contact_type, //H列: 連絡手段（メール）
    formObject.houseType, // I列: 住居タイプ (一軒家)
    fomrObject.house_info, //J列: 備考
    // 必要に応じてフォームのname属性と合わせて追加してください
  ];
  
  sheet.appendRow(rowData);
  
  return "お申し込みが完了しました！";

      function handleFormSubmit(formObject) {
        // 送信ボタンを無効化して二重送信を防ぐ
        document.getElementById('submitBtn').disabled = true;
        document.getElementById('submitBtn').innerText = "送信中...";
        document.getElementById('message').innerText = "";

        // GASの processForm 関数を呼び出す
        google.script.run
          .withSuccessHandler(function(msg) {
            document.getElementById('message').style.color = "green";
            document.getElementById('message').innerText = msg;
            document.getElementById('myForm').reset(); // フォームをクリア
            resetButton();
          })
          .withFailureHandler(function(err) {
            document.getElementById('message').style.color = "red";
            document.getElementById('message').innerText = "エラーが発生しました: " + err.message;
            resetButton();
          })
          .processForm(formObject);
      }

      function resetButton() {
        document.getElementById('submitBtn').disabled = false;
        document.getElementById('submitBtn').innerText = "送信する";
      }
    </script>
